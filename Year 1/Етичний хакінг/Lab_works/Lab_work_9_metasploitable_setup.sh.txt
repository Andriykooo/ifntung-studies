#!/bin/bash
# EGR6301 Lab 9 — Metasploitable 2 instructor setup
# Run as root on Metasploitable 2 BEFORE distributing the lab to students.
#
# What this does:
#   1. Installs a custom SUID-root binary at /usr/local/bin/find for Task 2.2
#   2. Adds a NOPASSWD sudoers rule on /usr/bin/awk for Tasks 3.1-3.2
#   3. Validates everything and prints the artifacts students should see.
#
# Usage on Metasploitable 2:
#   ssh msfadmin@<IP>
#   sudo su -
#   bash lab9_metasploitable_setup.sh
#
# To revert (after the lab):
#   rm /usr/local/bin/find
#   sed -i '/msfadmin ALL=(ALL) NOPASSWD: \/usr\/bin\/awk/d' /etc/sudoers

set -e

if [ "$(id -u)" -ne 0 ]; then
    echo "[!] This script must be run as root."
    exit 1
fi

echo "================================================================"
echo " EGR6301 Lab 9 — Metasploitable 2 setup"
echo "================================================================"
echo ""

# ---------------------------------------------------------------
# Task 2.2 — Custom SUID binary in a non-standard location
# ---------------------------------------------------------------
# Choice: copy /usr/bin/find to /usr/local/bin/find and set SUID.
# Rationale: /usr/local/bin is not part of base Debian Etch's SUID
# inventory, so students who scan for SUID files will see it stand
# out by location. find has a clean GTFOBins entry under SUID:
#   /usr/local/bin/find . -exec /bin/sh -p \; -quit
# which gives euid=0 (matching the lab's emphasis on euid=0 vs uid=0).

echo "[*] Task 2.2: installing custom SUID at /usr/local/bin/find ..."
cp /usr/bin/find /usr/local/bin/find
chown root:root /usr/local/bin/find
chmod 4755 /usr/local/bin/find

# ---------------------------------------------------------------
# Tasks 3.1–3.2 — NOPASSWD sudoers rule on a wide tool
# ---------------------------------------------------------------
# Choice: msfadmin ALL=(ALL) NOPASSWD: /usr/bin/awk
# Rationale: cleanest "NOPASSWD on a wide tool" pattern from the
# lecture. awk has a one-line GTFOBins entry:
#   sudo awk 'BEGIN {system("/bin/sh")}'
# Different binary from the SUID one (find), so Parts 2 and 3
# exercise distinct techniques rather than the same trick twice.

RULE="msfadmin ALL=(ALL) NOPASSWD: /usr/bin/awk"

echo "[*] Tasks 3.1-3.2: adding NOPASSWD sudoers rule ..."
if grep -qF "$RULE" /etc/sudoers; then
    echo "    (rule already present, skipping)"
else
    # Append via tee so the redirect happens before the pipe boundary
    echo "$RULE" >> /etc/sudoers
fi

echo "[*] Validating /etc/sudoers syntax ..."
visudo -c

# ---------------------------------------------------------------
# Verification block — what students will see
# ---------------------------------------------------------------
echo ""
echo "================================================================"
echo " Verification"
echo "================================================================"
echo ""
echo "[+] Custom SUID binary listing:"
ls -la /usr/local/bin/find
echo ""
echo "[+] sudoers entries for msfadmin:"
grep -n "msfadmin" /etc/sudoers
echo ""
echo "[+] Sanity check — find SUID surface as msfadmin would see it:"
sudo -u msfadmin find / -perm -4000 -type f 2>/dev/null | grep -E "(local|opt)" || true
echo ""
echo "[+] Sanity check — sudo -l output as msfadmin (no password):"
sudo -u msfadmin sudo -ln 2>/dev/null || true
echo ""
echo "================================================================"
echo " Setup complete. Lab 9 is ready for distribution."
echo "================================================================"
